def count_fingers(hand_landmarks, handedness="Right"):
    fingers = 0
    wrist = hand_landmarks[0]
    middle_mcp = hand_landmarks[9]
    hand_size = math.sqrt((middle_mcp.x - wrist.x)**2 + (middle_mcp.y - wrist.y)**2)

    finger_tips = [8, 12, 16, 20]
    finger_pips = [6, 10, 14, 18]
    
    for tip_idx, mcp_idx, pip_idx in zip(finger_tips, finger_pips, [7, 11, 15, 19]):
        tip = hand_landmarks[tip_idx]
        mcp = hand_landmarks[mcp_idx]
        pip = hand_landmarks[pip_idx]
        
        vertical_extension = mcp.y - tip.y
        mcp_pip_dist = math.sqrt((pip.x - mcp.x)**2 + (pip.y - mcp.y)**2)
        pip_tip_dist = math.sqrt((tip.x - pip.x)**2 + (tip.y - pip.y)**2)
        mcp_tip_dist = math.sqrt((tip.x - mcp.x)**2 + (tip.y - mcp.y)**2)
        
        normalized_vertical = vertical_extension / hand_size
        normalized_extension = mcp_tip_dist / hand_size
        
        is_extended = (normalized_vertical > 0.25 and normalized_extension > 0.35 and 
                      pip_tip_dist > mcp_pip_dist * 0.4)
        
        if is_extended:
            fingers += 1

    thumb_tip = hand_landmarks[4]
    thumb_ip = hand_landmarks[3]
    thumb_mcp = hand_landmarks[2]
    index_mcp = hand_landmarks[5]
    
    dx = thumb_tip.x - thumb_mcp.x
    dy = thumb_tip.y - thumb_mcp.y
    thumb_distance = math.sqrt(dx*dx + dy*dy)
    normalized_distance = thumb_distance / hand_size
    
    thumb_index_dist = math.sqrt((thumb_tip.x - index_mcp.x)**2 + (thumb_tip.y - index_mcp.y)**2)
    normalized_thumb_index = thumb_index_dist / hand_size
    
    v1_x = thumb_ip.x - thumb_mcp.x
    v1_y = thumb_ip.y - thumb_mcp.y
    v2_x = thumb_tip.x - thumb_mcp.x
    v2_y = thumb_tip.y - thumb_mcp.y
    
    dist_to_ip = math.sqrt(v1_x*v1_x + v1_y*v1_y)
    dist_to_tip = math.sqrt(v2_x*v2_x + v2_y*v2_y)
    ratio = dist_to_tip / (dist_to_ip + 0.001)
    
    conditions_met = 0
    if normalized_distance > 0.50:
        conditions_met += 1
    if normalized_thumb_index > 0.65:
        conditions_met += 1
    if ratio > 1.8:
        conditions_met += 1
    
    if conditions_met >= 2:
        fingers += 1
    
    return fingers

def smooth_finger_count(current_count):
    finger_buffer.append(current_count)
    if len(finger_buffer) < SMOOTHING_FRAMES:
        return current_count
    counts = Counter(finger_buffer)
    return counts.most_common(1)[0][0]

def map_gesture_to_command(fingers):
    command_map = {
        0: ("L0;F0;A0;D0", "All OFF"),
        1: ("L1", "Light ON"),
        2: ("D1", "Door Unlock"),
        3: ("A1", "AC ON"),
        4: ("F1", "Fan ON"),
        5: ("LD1", "LCD ON"),
    }
    return command_map.get(fingers, (None, "Unknown"))

def recognize_face(frame):
    if face_model is None:
        return None, 0.0

    small_frame = cv2.resize(frame, (320, 240))
    gray = cv2.cvtColor(small_frame, cv2.COLOR_BGR2GRAY)
    
    faces = face_cascade.detectMultiScale(gray, scaleFactor=1.1, minNeighbors=5, minSize=(50, 50))

    if len(faces) == 0:
        return None, 0.0

    scale_x = frame.shape[1] / 320
    scale_y = frame.shape[0] / 240
    faces_scaled = [(int(x*scale_x), int(y*scale_y), int(w*scale_x), int(h*scale_y)) 
                    for (x, y, w, h) in faces]
    
    faces_sorted = sorted(faces_scaled, key=lambda rect: rect[2] * rect[3], reverse=True)
    x, y, w, h = faces_sorted[0]

    if USE_GRAYSCALE:
        gray_full = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        face_roi = gray_full[y:y+h, x:x+w]
        face_resized = cv2.resize(face_roi, (224, 224))
        face_normalized = face_resized.astype(np.float32) / 255.0
        face_normalized = np.expand_dims(face_normalized, axis=-1)
    else:
        face_roi = frame[y:y+h, x:x+w]
        face_resized = cv2.resize(face_roi, (224, 224))
        face_rgb = cv2.cvtColor(face_resized, cv2.COLOR_BGR2RGB)
        face_normalized = face_rgb.astype(np.float32) / 255.0
    
    face_batch = np.expand_dims(face_normalized, axis=0)
    prediction = face_model.predict(face_batch, verbose=0, batch_size=1)
    predicted_idx = np.argmax(prediction)
    confidence = float(np.max(prediction))
    predicted_name = FACE_CLASS_NAMES[predicted_idx]

    color = (0, 255, 0) if confidence > CONFIDENCE_THRESHOLD else (0, 0, 255)
    cv2.rectangle(frame, (x, y), (x+w, y+h), color, 2)
    cv2.putText(frame, f"{predicted_name}", (x, y-10),
                cv2.FONT_HERSHEY_SIMPLEX, 0.6, color, 2)
    cv2.putText(frame, f"{confidence:.2f}", (x, y+h+20),
                cv2.FONT_HERSHEY_SIMPLEX, 0.5, color, 2)

    return predicted_name, confidence

def send_command(cmd, source=""):
    try:
        arduino.reset_output_buffer()
        arduino.write((cmd + '\n').encode())
        arduino.flush()
        time.sleep(0.05)
        
        print(f"[PYTHON] Sent: {cmd} ({source})")
        
    except Exception as e:
        print(f"[PYTHON] ✗ Send Error: {e}")
